cmake_minimum_required(VERSION 3.24)
project(pixeltrace_llama LANGUAGES C CXX)

if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "PixelTrace llama bridge requires Android arm64-v8a")
endif()
if(NOT DEFINED LLAMA_SOURCE_DIR)
    message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned llama.cpp source checkout")
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
find_package(SPIRV-Headers CONFIG REQUIRED)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
# The pinned upstream finds the header package but does not link its interface
# target. Explicitly propagate the installed SPIR-V include directory.
target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)

add_library(pixeltrace_llama SHARED PixelTraceLlama.cpp)
target_compile_features(pixeltrace_llama PRIVATE cxx_std_17)
target_compile_definitions(pixeltrace_llama PRIVATE PT_LLAMA_REVISION="c0fd78626405c6d9b934efef0ec7291fcc20e3d6")
target_link_libraries(pixeltrace_llama PRIVATE mtmd llama ggml log)
target_link_options(pixeltrace_llama PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
set_target_properties(pixeltrace_llama PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
